        ```js
        var items = event.message.cart_items;

                currency: event.fullState.default.site.currency,
                content_type: 'product'
            }
        ```

        ```js
        const event_id = window._satellite.cookie.get('eventId') || '(not-set)';
            const items = event.message.cart_items;
            const orderDetails = event.message.order;

            const uemHash = event.fullState.default.user.uem_hashed;

            if (items && orderDetails && window.ttq) {
                const formatProduct = (productArray = []) => {
                    return productArray.map((item) => ({
                            content_id: item.child_sku,
                            content_name: item.name,
                            quantity: item.qty,
                            price: item.listed_price,
                            content_type: 'product',
                        }));
                    };
            
                ttq.track('CompletePayment', {
                    contents: formatProduct(items),
                    value: orderDetails.total,
                    currency: 'AUD',
                },{ event_id });

                if (uemHash) {
                ttq.identify({
                    sha256_email: uemHash,
                })
                }
            
            _satellite.logger.info('✅ TikTok Purchase Event');
            }
        ```

        ```js
        const items = event.message?.cart_items;
        const order = event.message?.order;
        const error = order?.error;

        const customerStatus = window._satellite.cookie.get('order_event_id') ? 'return' : 'new';

        if (items && order && !error) {
            ((a, b, c) => { a[b] = a[b] || function () { (a[b].q = a[b].q || []).push(arguments); }; a[c] = a[b]; })(window, 'CommissionFactory', 'cf');

            items.forEach((item) => {
                cf('add', 'items', {
                    sku: item.child_sku,
                    quantity: item.qty,
                    price: parseFloat(item.sale_price).toFixed(2),
                });
            });

            if(order?.promo) { cf('set', 'coupon', order.promo[0].code ?? order.promo.code) }
            cf('set', 'order', order.orderId);
            cf('set', 'amount', parseFloat(order.revenue).toFixed(2));
            cf('set', 'currency', order.currency); 
            cf('set', 'customer', customerStatus);
            cf('track');

            _satellite.logger.info('🏭 Commission Factory: Track Purchase');
        };
        ```
