        ```js
        const item = event.message?.product?.[0] ?? event.message?.products?.[0];

            const errorLog = [];
            const error = (prop) => { errorLog.push(prop); return undefined; };

            const formatProduct = (product = {}) => {
                const listedPrice = product.listed_price ?? error("listed_price");

                return { 
                    product_name: product.name ?? error("product_name"),
                    product_id: product.child_sku ?? error("product_id"),
                    listed_price: listedPrice,
                    on_sale: product.is_markdown ?? error("on_sale"),
                    gender: product.gender ?? error('gender'),
                    brand: product.brand ?? error("brand"),
                    parent_category: product.parent_category ?? error("parent_category"),
                    category: product.category ?? error("category"),
                    color: product.color ?? error("color"),
                    quantity: 1,
                    item_total: listedPrice,
                    $value: listedPrice,
                    $extra: {
                        in_stock: product.sku_available ?? error("in_stock"),
                        parent_sku: product.parent_sku ?? error("parent_sku"),
                        full_price: product.full_price ?? error("full_price"),
                        url: event.message.default.page.url ?? error("url"),
                        currency: event.fullState.default.site.currency ?? error("currency")
                    }
                }
            }

            const payload = formatProduct(item);

            if (errorLog.length !== 0) {
                _satellite.track('action_failed', {  
                    rule: 'analytics::view_tracking::product_view', 
                    action: 'send_product_view_to_klaviyo', 
                    message: 'Fields Missing from datalayer', 
                    status_code: 422,
                    missing_fields: errorLog 
                });
                _satellite.logger.error("📨 Klaviyo: Error ❌\n Missing fields:", errorLog);
            }

            if (window.klaviyo) { 
                klaviyo.track("Viewed Product", payload);
                _satellite.logger.info("📨 Klaviyo: Viewed product Sent ✔️\n", payload);
            } else {
                _satellite.track('action_failed', {  
                    rule: 'analytics::view_tracking::product_view', 
                    action: 'send_product_view_to_klaviyo', 
                    message: 'klaviyo not available',
                    status_code: 424 
                });
                _satellite.logger.error("📨 Klaviyo: Error ❌\n window.klaviyo unavailable, check that script loaded correctly");
            }
        ```
