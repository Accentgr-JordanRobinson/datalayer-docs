        const dl = event.message;
        const items = dl.cart_items ?? [];

        const promoBase = Number(dl.order?.promo?.[0]?.base) || 0;
        const promoGst = Number(dl.order?.promo?.[0]?.gst) || 0;
        const totalDiscount = promoBase + promoGst;

        function formatProduct(product) {
        if (!product) return {};
        
        return {
            ...(product.name && { item_name: product.name }),
            ...(product.child_sku && { item_id: product.child_sku }),
            ...(product.parent_sku && { item_sku: product.parent_sku }),
            ...(product.brand && { item_brand: product.brand }),
            ...(product.parent_category && { item_category: product.parent_category }),
            ...(product.gender && { item_category2: product.gender }),
            ...(product.sport && { item_category3: product.sport }),
            ...(product.specialty && { item_category4: product.specialty }),
            ...(product.model && { item_category5: product.model }),
            ...((product.size || product.color) && { 
            item_variant: [
                product.size, 
                product.color
            ].filter(Boolean).join(' ') 
            }),
            ...(product.listed_price && { price: Number(product.listed_price) }),
            ...(product.full_price && product.listed_price && { 
            discount: Number((product.full_price - product.listed_price).toFixed(2))
            }),
            ...(product.qty && { quantity: Number(product.qty) }),
            ...(product.position && { index: product.position }),
        };
        }

        function applyPromoDiscount(formattedItems, discount) {
        if (discount <= 0) return formattedItems;
        
        const totalQuantity = formattedItems.reduce((sum, item) => sum + (item.quantity || 0), 0);
        
        if (totalQuantity === 0) return formattedItems;
        
        const perItemDiscount = Number((discount / totalQuantity).toFixed(2));
        
        return formattedItems.map(item => ({
            ...item,
            discount: (item.discount || 0) + perItemDiscount
        }));
        }

        let formattedItems = items.map(formatProduct);
        formattedItems = applyPromoDiscount(formattedItems, totalDiscount);

        gtag("event", "purchase", {
        send_to: [ dl.default?.site?.domain === 'www.dickiesanz.com' ? 'G-26TFT5F1H8' : 'G-20EDY3BXWH'],
        event_category: "ecommerce",
        ...(dl.order?.total && { value: Number(dl.order.total) }),
        currency: dl.default?.site?.currency,
        items: formattedItems,
        ...(dl.order?.tax && { tax: Number(dl.order.tax) }),
        ...(dl.order?.shipping?.base && { shipping: Number(dl.order.shipping.base) }),
        ...(dl.order?.promo && { coupon: dl.order.promo[0].code ?? dl.order.promo.code }),
        ...(dl.order?.orderId && { transaction_id: dl.order.orderId }),
        ...(dl.order?.payments?.[0].type && { payment_type: dl.order.payments[0].type }),
        ...(dl.order?.delivery?.method && { shipping_tier: dl.order.delivery.method }),
        });
