const event_name = getDataElementValue('dl_event_name');

const event_map = {
  "default_view": "PageView",
  "page_default": "PageView",
  "home_view": "PageView",
  "product_listing-view": "PageView", 
  "product_view": "ViewContent",
  "cart_add": "AddToCart",
  "checkout_start": "InitiateCheckout",
  "order_success": "Purchase"
};

return event_map[event_name];
