# Default Module

The `default` module contains all the core information for the app. This is further broken down into `site`, `page` & `user` data.

>[!INFO]
>The default data needs to be pushed to the data layer whenever any of the following conditions are met;
>1. The app has been loaded for the first time (ie. The user opens the website in their browser marking the start of their session).
>2. The DOM has been rerendered (ie. The application is being rendered as a Static Site & the user navigates to a page triggering a page reload).
>3. Any information has changed  (ie. If a user navigates from 'Home' page to 'Products' page the `page.type` will be updated from "home" to "product").

    interface Site {
      name: string;
      experience: 'desktop' | 'mobile' | 'tablet';
      currency: string;
      division: string;
      domain: string;
      env: 'dev' | 'stg' | 'prod';
      version: string;
    }

    interface Page {
      type: string;
      action: string;
      path: string;
      title: string;
      url: string;
    }

    interface User {
      user_state: 'guest' | 'customer';
      login_state: 'logged-in' | 'anonymous';
      uem_hashed?: string;
      session_id?: string;
      division_id?: string;
    }

    interface Default {
        event: 'page_default';
        default: {
            site: Site;
            page: Page;
            user: User;
        }
    }

    // Example Event
    const default: Default = {
