                if (userData && userData.uem_hashed) {

                    return
                    }
                    
                    // set hashed email
                    window._satellite.cookie.set('default.user.uem_hashed', userData.uem_hashed);

                    if(window.gtag){

                    }
                    
                    // set erp_key
                    if(userData.division_id!= ""){
                    window._satellite.cookie.set('default.user.division_id', userData.division_id);

                    logger('Set User Variables');
                }

            var logger = _satellite.logger.info

            // Retrieve the stored hash
            var storedHash = window._satellite.getVar('ls.default.user.uem_hashed');

            // Get the new hash from the event
            var newHash = event?.message?.default?.user?.uem_hashed;

            // Validate and process the new hash
            var userData;
            if (newHash && newHash !== "" && newHash !== storedHash) {
                setUserVariables( event.message.default.user );
